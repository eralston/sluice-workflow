// Reviewer probe (E4a): every first merge on both pods of pod-sprint, played at 1x; then a seek mid-merge and a pause mid-merge on each.
export default async function probe(d: any): Promise<void> {
  const { page } = d;
  await page.locator('.callout--galaxy', { hasText: 'Tidewater Array' }).first().click();
  await page.locator('.chooser button', { hasText: 'Sprint 13' }).click();
  await page.waitForFunction(() => (window as any).__starChartTest!.viewGraph()?.sceneKey.startsWith('sprint:'), undefined, { timeout: 20_000 });
  await d.settle();
  for (const pod of ['Harbor Pod', 'Lantern Pod']) {
    const inPod = await page.evaluate(() => (window as any).__starChartTest!.viewGraph()?.sceneKey.startsWith('pod:'));
    if (inPod) { await page.keyboard.press('Escape'); await page.waitForTimeout(1500); await d.settle(); }
    await page.locator('.callout', { hasText: pod }).first().dblclick();
    await page.waitForFunction(() => (window as any).__starChartTest!.viewGraph()?.sceneKey.startsWith('pod:'), undefined, { timeout: 20_000 });
    await d.settle();
    const merges = await page.evaluate(async () => {
      const t = (window as any).__starChartTest!;
      const frame = () => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r)));
      const mg = (stage: string) => Number(/\|mg=(\d+)/.exec(stage)?.[1] ?? 0);
      let before = new Map<string, number>();
      const out: { startMs: number; label: string }[] = [];
      for (const b of t.beatTimes()) {
        t.setScheduleMs(b.startMs + 1);
        await frame();
        const fac = t.viewGraph()!.nodes.filter((n: any) => n.kind === 'facility');
        for (const n of fac) { const was = before.get(n.projectionKey); if (was === 0 && mg(n.stage) > 0) out.push({ startMs: b.startMs, label: n.label }); }
        before = new Map(fac.map((n: any) => [n.projectionKey, mg(n.stage)]));
      }
      return out;
    });
    d.note(`${pod} first merges: ${JSON.stringify(merges)}`);
    for (const m of merges) {
      const tag = `${pod.split(' ')[0]}-${m.label}`;
      // 1) forward play
      d.note(`PLAY ${tag} at ${m.startMs}`);
      await page.evaluate(() => { const t = (window as any).__starChartTest!; t.setReduced(false); t.setRate(1); });
      await page.evaluate((x: number) => (window as any).__starChartTest!.setScheduleMs(x - 1), m.startMs);
      await page.waitForTimeout(400);
      await page.evaluate(() => (window as any).__starChartTest!.stepBeat());
      await page.waitForTimeout(2000); await d.shot(`${tag}-play-2s`);
      await page.waitForTimeout(2200); await d.shot(`${tag}-play-4.2s`);
      await page.waitForTimeout(5000);
      d.note(`END PLAY ${tag}`);
      // 2) seek mid-merge (1.5 s in): back to just before the beat
      d.note(`SEEK ${tag}`);
      await page.evaluate((x: number) => (window as any).__starChartTest!.setScheduleMs(x - 1), m.startMs);
      await page.waitForTimeout(400);
      await page.evaluate(() => (window as any).__starChartTest!.stepBeat());
      await page.waitForTimeout(1500);
      await page.evaluate((x: number) => (window as any).__starChartTest!.setScheduleMs(x - 1), m.startMs);
      await page.waitForTimeout(800); await d.shot(`${tag}-after-seek-back`);
      await page.waitForTimeout(5000); await d.shot(`${tag}-after-seek-back-5s`);
      d.note(`END SEEK ${tag}`);
      // 3) pause mid-merge (1.0 s in), hold 6 s
      d.note(`PAUSE ${tag}`);
      await page.evaluate(() => (window as any).__starChartTest!.stepBeat());
      await page.waitForTimeout(1000);
      await page.evaluate(() => (window as any).__starChartTest!.pause());
      await page.waitForTimeout(6000); await d.shot(`${tag}-paused-6s`);
      await page.evaluate(() => (window as any).__starChartTest!.play());
      await page.waitForTimeout(8000);
      await page.evaluate(() => (window as any).__starChartTest!.pause());
      d.note(`END PAUSE ${tag}`);
    }
  }
}
